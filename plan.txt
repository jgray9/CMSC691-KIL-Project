def convert-to-prompt(s1, s2):
    combine sentences and answer into prompt and return
    """Sentence 1: {s1}
    Sentence 2: {s2}
    Do these sentences mean the same thing? Write 'Yes' if they do, and 'No' if they don't."""

def augment-wordnet(s):
    nouns = list of nouns in s
    if nouns empty return s
    for n in nouns:
        synonyms = list of synonyms for n
        if no synonyms continue
        replace first instance of n in s
    return s

def augment-lesk(s):
    nouns = list of nouns in s
    if nouns empty return s
    for n in nouns:
        synonyms = list of synonyms for n using lesk
        if no synonyms continue
        replace first instance of n in s
    return s

def augment-bert(s):
    nouns = list of nouns in s
    if nouns empty return s
    for n in nonus:
        replace first instance of n in s with MASK
        preds = predictions from BERT
        replace MASK with highest rated prediction
    return s

def augment-hybrid(s):
    nouns = list of nouns in s
    if nouns empty return s
    for n in nonus:
        synonyms = list of synonyms for n
        if no synonyms continue
        preds = predictions from BERT
        replace MASK with highest pred that is in synonyms
    return s
