def query-wordnet(word):
    synsets = query wordnet
    lemmas = combine lemmas from each set
    syns = set of synonyms = name of lemmas
    remove word from syns
    return syns

def query-lesk(word, sen):
    synset = run lesk
    lemmas = lemmas from synset
    syns = set of synonyms = name of lemmas
    remove word from syns
    return syns

def query-bert(word, sen):
    replace word in sen with [MASK]
    inputs = tokenized sen
    mask_idx = index of mask token in input tokens tensor
    logits = run bert on inputs
    logits = view only logits for mask_idx position

    return dict{
        word: decoded token,
        score: logit score
    }

def augment-wordnet(sen):
    nouns = list of nouns in sen
    if nouns empty return sen
    for n in nouns:
        syn = query-wordnet(n)
        if no synonyms continue
        replace first instance of n with syn
    return sen

def augment-lesk(sen):
    nouns = list of nouns in sen
    if nouns empty return sen
    for n in nouns:
        synonyms = query-lesk(n, sen)
